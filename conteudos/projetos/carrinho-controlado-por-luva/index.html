transmissor-luva.ino

Baixar arquivo 
#include <WiFi.h>
#include <esp_now.h>

const int PIN_EIXO_X = 34;
const int PIN_EIXO_Y = 35;
const int PIN_BOTAO = 23;

uint8_t macDoCarrinho[] = {0x24, 0x6F, 0x28, 0xAA, 0xBB, 0xCC};

struct PacoteControle {
  int16_t eixoX;
  int16_t eixoY;
  bool botaoPressionado;
};

void loop() {
  PacoteControle pacote;
  pacote.eixoX = map(analogRead(PIN_EIXO_X), 0, 4095, -100, 100);
  pacote.eixoY = map(analogRead(PIN_EIXO_Y), 0, 4095, -100, 100);
  pacote.botaoPressionado = digitalRead(PIN_BOTAO) == LOW;

  esp_now_send(macDoCarrinho, (uint8_t *)&pacote, sizeof(pacote));
  delay(80);
}

#include <WiFi.h>
#include <esp_now.h>

struct PacoteControle {
  int16_t eixoX;
  int16_t eixoY;
  bool botaoPressionado;
};

PacoteControle ultimoComando = {0, 0, false};

void moverCarrinho(int eixoX, int eixoY) {
  int velocidadeBase = map(abs(eixoY), 0, 100, 0, 255);
  int ajusteCurva = map(eixoX, -100, 100, -120, 120);

  int esquerda = velocidadeBase - ajusteCurva;
  int direita = velocidadeBase + ajusteCurva;

  if (eixoY < 0) {
    esquerda *= -1;
    direita *= -1;
  }

  controlarMotor(MOTOR_ESQ_IN1, MOTOR_ESQ_IN2, MOTOR_ESQ_PWM, esquerda);
  controlarMotor(MOTOR_DIR_IN1, MOTOR_DIR_IN2, MOTOR_DIR_PWM, direita);
}

Funcionamento

Como o projeto funciona

O objetivo é transformar o movimento da mão em comandos de direção para o carrinho.

1

A luva lê o joystick

O ESP32 transmissor mede os eixos X e Y e transforma os valores em uma escala simples de -100 a 100.
2

Os dados viajam por ESP-NOW

O pacote de controle é enviado diretamente para o ESP32 do carrinho, sem depender de roteador Wi-Fi.
3

O carrinho calcula o movimento

O eixo Y define frente ou ré. O eixo X ajusta a diferença de velocidade entre os motores para fazer curvas.

Conceitos trabalhados

  • Entrada analógica

  • PWM e controle de velocidade

  • Mostrar status em LEDs coloridos.
  • Trocar joystick por sensor de inclinação.

2 ESP32, joystick, botão, driver de motor, 2 motores DC, chassi, rodas, bateria e fios jumper.

Cuidados importantes

Confira a polaridade da bateria, mantenha GND comum e nunca alimente motores diretamente pelos pinos do ESP32.
